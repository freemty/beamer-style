\section{Mathematics}

\begin{frame}{Equations and Theorems}
  \begin{theorem}[Sample]
    For all $n \geq 1$:
    \[
      \sum_{k=1}^{n} k = \frac{n(n+1)}{2}
    \]
  \end{theorem}

  \begin{proof}
    By induction on $n$. Base case $n=1$ is trivial.
  \end{proof}
\end{frame}
